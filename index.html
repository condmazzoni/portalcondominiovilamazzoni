import express from 'express'; import cors from 'cors'; import rateLimit from 'express-rate-limit'; import twilio from 'twilio'; import nodemailer from 'nodemailer'; import fs from 'fs'; import path from 'path'; import { fileURLToPath } from 'url'; const __dirname = path.dirname(fileURLToPath(import.meta.url)); const DB_PATH = path.join(__dirname, 'data.json'); const app = express(); app.use(express.json({ limit: '8mb' })); app.use(cors({ origin: process.env.ALLOWED_ORIGIN || '*' })); // Limite básico de uso: no máximo 120 requisições por minuto por IP. // (mais alto que antes porque agora o painel inteiro depende deste servidor // para ler e salvar dados, não só para enviar mensagens) const limiter = rateLimit({ windowMs: 60 * 1000, max: 120 }); app.use(limiter); function checkApiKey(req, res, next) { const key = req.header('x-api-key'); if (!process.env.API_KEY || key !== process.env.API_KEY) { return res.status(401).json({ erro: 'Chave de API ausente ou inválida.' }); } next(); } /* ---------- banco de dados simples (arquivo JSON no disco) ---------- Guarda tudo do painel: condomínios, moradores, mensagens, visitantes, configurações. Não é um banco de verdade, mas resolve o problema de persistência sem exigir contratar outro serviço. ATENÇÃO: no plano gratuito do Render, esse arquivo pode ser apagado quando o backend for republicado (um novo "deploy"). Para uso pesado ou definitivo, o próximo passo seria trocar isso por um banco de dados de verdade (ex: PostgreSQL). */ let db = {}; try { if (fs.existsSync(DB_PATH)) { db = JSON.parse(fs.readFileSync(DB_PATH, 'utf8')); } } catch (e) { console.error('Não foi possível ler data.json, começando vazio.', e.message); db = {}; } function salvarDb() { try { fs.writeFileSync(DB_PATH, JSON.stringify(db), 'utf8'); } catch (e) { console.error('Não foi possível gravar data.json:', e.message); } } app.post('/api/kv/get', checkApiKey, (req, res) => { const { key } = req.body || {}; if (!key) return res.status(400).json({ erro: 'key é obrigatório.' }); const valor = Object.prototype.hasOwnProperty.call(db, key) ? db[key] : null; res.json(valor === null ? null : { key, value: valor }); }); app.post('/api/kv/set', checkApiKey, (req, res) => { const { key, value } = req.body || {}; if (!key) return res.status(400).json({ erro: 'key é obrigatório.' }); db[key] = value; salvarDb(); res.json({ key, value }); }); app.post('/api/kv/delete', checkApiKey, (req, res) => { const { key } = req.body || {}; if (!key) return res.status(400).json({ erro: 'key é obrigatório.' }); delete db[key]; salvarDb(); res.json({ key, deleted: true }); }); app.post('/api/kv/list', checkApiKey, (req, res) => { const { prefix } = req.body || {}; const keys = Object.keys(db).filter(k => !prefix || k.startsWith(prefix)); res.json({ keys }); }); const twilioClient = process.env.TWILIO_ACCOUNT_SID && process.env.TWILIO_AUTH_TOKEN ? twilio(process.env.TWILIO_ACCOUNT_SID, process.env.TWILIO_AUTH_TOKEN) : null; const mailer = process.env.GMAIL_USER && process.env.GMAIL_APP_PASSWORD ? nodemailer.createTransport({ service: 'gmail', auth: { user: process.env.GMAIL_USER, pass: process.env.GMAIL_APP_PASSWORD } }) : null; async function enviarWhatsapp(paraNumero, texto) { if (!twilioClient) return { enviado: false, erro: 'Twilio não configurado no servidor.' }; if (!paraNumero) return { enviado: false, erro: 'Número de destino não informado.' }; try { let numero = paraNumero.replace(/[^\d+]/g, ''); if (!numero.startsWith('+')) numero = '+55' + numero; // assume Brasil se não vier com DDI await twilioClient.messages.create({ from: process.env.TWILIO_WHATSAPP_FROM, to: 'whatsapp:' + numero, body: texto }); return { enviado: true }; } catch (e) { return { enviado: false, erro: e.message }; } } async function enviarEmail(paraEmail, assunto, texto, anexo) { if (!mailer) return { enviado: false, erro: 'E-mail não configurado no servidor.' }; if (!paraEmail) return { enviado: false, erro: 'E-mail de destino não informado.' }; try { const opts = { from: process.env.GMAIL_USER, to: paraEmail, subject: assunto || 'Portal do condomínio', text: texto }; if (anexo && anexo.dataUrl) { const match = /^data:(.+);base64,(.+)$/.exec(anexo.dataUrl); if (match) { opts.attachments = [{ filename: anexo.nome || 'anexo', content: Buffer.from(match[2], 'base64'), contentType: match[1] }]; } } await mailer.sendMail(opts); return { enviado: true }; } catch (e) { return { enviado: false, erro: e.message }; } } app.get('/', (req, res) => { res.json({ status: 'ok', servico: 'backend do portal do condomínio' }); }); // Notifica o síndico (usado quando um morador envia mensagem ou cadastra visitante) app.post('/api/notificar-sindico', checkApiKey, async (req, res) => { const { canalWhatsapp, canalEmail, whatsappPara, emailPara, assunto, mensagem, anexo } = req.body || {}; if (!mensagem) return res.status(400).json({ erro: 'Campo "mensagem" é obrigatório.' }); const resultado = {}; if (canalWhatsapp) resultado.whatsapp = await enviarWhatsapp(whatsappPara, mensagem); if (canalEmail) resultado.email = await enviarEmail(emailPara, assunto, mensagem, anexo); res.json(resultado); }); // Envia um comunicado do síndico para uma lista de moradores selecionados app.post('/api/enviar-broadcast', checkApiKey, async (req, res) => { const { mensagem, canalWhatsapp, canalEmail, destinatarios, assunto } = req.body || {}; if (!mensagem) return res.status(400).json({ erro: 'Campo "mensagem" é obrigatório.' }); if (!Array.isArray(destinatarios) || destinatarios.length === 0) { return res.status(400).json({ erro: 'Lista de destinatários vazia ou inválida.' }); } const resultados = []; for (const d of destinatarios) { const item = { nome: d.nome }; if (canalWhatsapp) item.whatsapp = await enviarWhatsapp(d.whatsapp, mensagem); if (canalEmail) item.email = await enviarEmail(d.email, assunto, mensagem); resultados.push(item); } res.json({ resultados }); }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => console.log('Servidor rodando na porta ' + PORT));
